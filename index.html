// === KODE GOOGLE APPS SCRIPT (SUPPORTS BACA & TULIS DATA) ===
function doGet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var members = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][1]) {
      members.push({
        id: String(data[i][0] || ("ADM" + i)),
        name: String(data[i][1]),
        store: String(data[i][2] || "Toko A"),
        phone: String(data[i][3] || "")
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "success", members: members }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var body = JSON.parse(e.postData.contents);
  if (body.action === "updateMembers") {
    sheet.clear();
    sheet.appendRow(["ID", "Nama Admin", "Toko", "No WA"]);
    body.members.forEach(function(m) {
      sheet.appendRow([m.id, m.name, m.store, m.phone || ""]);
    });
  }
  return ContentService.createTextOutput(JSON.stringify({status: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
